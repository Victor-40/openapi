import openai

# Настройка локального сервера
client = openai.Client(
    base_url="http://localhost:8088/v1",  # Порт зависит от конфигурации
    api_key="ollama",  # Для llama-server API key часто не нужен или можно указать "ollama"
)

# Запрос к модели
chat = client.chat.completions.create(
    model="Qwen3.5-4B-Q4_K_M",  # Или любую модель, загруженную в llama-server
    messages=[{"role": "user", "content": "какие параметры при запуске сервера для использования видеокарты intel?"}],
    temperature=0.7
)
print(chat.choices[0].message.content)
